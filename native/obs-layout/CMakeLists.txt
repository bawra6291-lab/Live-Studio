cmake_minimum_required(VERSION 3.28)
project(LiveDeskObsLayout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets)
add_library(live-desk-layout SHARED plugin.cpp)
target_link_libraries(live-desk-layout PRIVATE Qt6::Core Qt6::Widgets)
target_compile_definitions(live-desk-layout PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(layout-fixture fixture.cpp plugin.cpp)
target_link_libraries(layout-fixture PRIVATE Qt6::Core Qt6::Widgets)
target_compile_definitions(layout-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN LIVE_DESK_LAYOUT_FIXTURE)
